/**
 * YouTubeのプレイリストURL（または再生リストID）から、
 * 含まれる動画のタイトルとURLを全件取得する。
 * input: string（プレイリストのURL、もしくはIDそのもの）
 * 戻り値: [{ url, title }]
 *
 * 利用には、エディタ左メニュー「サービス」から
 * 「YouTube Data API」を追加しておく必要がある。
 */
function fetchPlaylistVideos(input) {
  const playlistId = extractPlaylistId_(input);
  if (!playlistId) {
    throw new Error('プレイリストのURLからIDを読み取れませんでした。URLを確認してください。');
  }

  const videos = [];
  let pageToken = null;

  do {
    let response;
    try {
      response = YouTube.PlaylistItems.list('snippet', {
        playlistId: playlistId,
        maxResults: 50,
        pageToken: pageToken || undefined
      });
    } catch (e) {
      // YouTube Data APIが未有効、プレイリストが非公開・存在しない等
      throw new Error(
        'プレイリストの取得に失敗しました： ' + (e && e.message ? e.message : e) +
        '　（エディタの「サービス」に「YouTube Data API」が追加されているか確認してください）'
      );
    }

    if (!response || !response.items) break;

    response.items.forEach(function (item) {
      const videoId = item.snippet && item.snippet.resourceId && item.snippet.resourceId.videoId;
      const videoTitle = item.snippet && item.snippet.title;
      // 削除済み・非公開動画はタイトルがこの文字列になることがあるのでスキップする
      if (!videoId || !videoTitle || videoTitle === 'Private video' || videoTitle === 'Deleted video') {
        return;
      }
      videos.push({
        url: 'https://youtu.be/' + videoId,
        title: videoTitle
      });
    });

    pageToken = response.nextPageToken;
  } while (pageToken);

  if (videos.length === 0) {
    throw new Error('プレイリストの中に動画が見つかりませんでした（非公開・削除済みのみの可能性があります）。');
  }

  return videos;
}

/**
 * YouTubeのプレイリストURLから再生リストIDを取り出す。
 * 対応例：
 *  https://www.youtube.com/playlist?list=PLxxxxxxxx
 *  https://www.youtube.com/watch?v=xxxx&list=PLxxxxxxxx
 *  PLxxxxxxxx （IDそのもの）
 */
function extractPlaylistId_(input) {
  const trimmed = (input || '').trim();
  if (!trimmed) return '';

  const match = trimmed.match(/[?&]list=([a-zA-Z0-9_-]+)/);
  if (match) return match[1];

  // URLではなく、IDそのものが貼られた場合
  if (/^[a-zA-Z0-9_-]{10,}$/.test(trimmed)) return trimmed;

  return '';
}

/**
 * YouTubeのURLから動画タイトル（＝出演者名として使う）を取得する
 * urls: string[]
 * 戻り値: [{ url, title, error }]
 */
function fetchYoutubeTitles(urls) {
  return (urls || []).map(function (url) {
    const trimmed = (url || '').trim();
    if (!trimmed) {
      return { url: trimmed, title: '', error: 'URLが空です' };
    }
    try {
      const oembedUrl = 'https://www.youtube.com/oembed?url=' + encodeURIComponent(trimmed) + '&format=json';
      const res = UrlFetchApp.fetch(oembedUrl, { muteHttpExceptions: true });
      if (res.getResponseCode() !== 200) {
        return { url: trimmed, title: '', error: 'タイトルを取得できませんでした（URLを確認してください）' };
      }
      const data = JSON.parse(res.getContentText());
      return { url: trimmed, title: data.title || '', error: null };
    } catch (e) {
      return { url: trimmed, title: '', error: 'タイトルを取得できませんでした（URLを確認してください）' };
    }
  });
}

/**
 * 「ネタ見せちゃん」フォルダを取得する（なければ作成する）
 */
function getOrCreateNetamiseFolder_() {
  const FOLDER_NAME = 'ネタ見せちゃん';
  const folders = DriveApp.getFoldersByName(FOLDER_NAME);
  if (folders.hasNext()) {
    return folders.next();
  }
  return DriveApp.createFolder(FOLDER_NAME);
}

/**
 * メイン処理：フォームを実際に作成する
 * payload = { title: string, description: string, performers: [{name, url}] }
 * 戻り値 = { editUrl, viewUrl, title }
 */
function createNetamiseForm(payload) {
  const title = (payload && payload.title || '').trim();
  const description = (payload && payload.description || '').trim();
  const performers = (payload && payload.performers || []).filter(function (p) {
    return p && p.name && p.name.trim();
  });

  if (!title) {
    throw new Error('フォームのタイトルが空です。開催日を入力してください。');
  }
  if (performers.length === 0) {
    throw new Error('出演者が1組も入力されていません。');
  }

  let step = '初期化';
  try {
    // 1. フォーム本体を作成
    step = 'フォームの作成（FormApp.create）';
    const form = FormApp.create(title);

    step = '説明文の設定';
    form.setDescription(description);

    step = 'ログイン要件の設定';
    try { form.setRequireLogin(false); } catch (e) { /* 個人アカウントでは無視してよい */ }

    step = 'メール収集設定';
    try { form.setCollectEmail(false); } catch (e) { /* 個人アカウントでは無視してよい */ }

    // 2. 出演者ごとに項目を追加
    performers.forEach(function (p) {
      const name = p.name.trim();
      const url = (p.url || '').trim();

      step = '「' + name + '」の動画項目を追加';
      try {
        if (url) {
          const videoItem = form.addVideoItem();
          videoItem.setTitle(name);
          videoItem.setVideoUrl(url);
          videoItem.setAlignment(FormApp.Alignment.CENTER);
          videoItem.setWidth(320);
        } else {
          form.addSectionHeaderItem().setTitle(name);
        }
      } catch (e) {
        // 動画として追加できない場合は、見出し＋URLのテキストに切り替えて続行する
        form.addSectionHeaderItem()
          .setTitle(name)
          .setHelpText(url ? ('動画リンク：' + url) : '');
      }

      step = '「' + name + '」の感想欄を追加';
      form.addParagraphTextItem()
        .setTitle('「' + name + '」へのコメント')
        .setRequired(false);
    });

    // 3. 次回のために出演者リストを保存
    step = '出演者リストの保存';
    try { saveLastPerformers_(performers); } catch (e) { /* 保存に失敗しても作成自体は続行する */ }

    step = 'フォルダへの移動';
    let folderMoveError = null;
    try {
      const folder = getOrCreateNetamiseFolder_();
      DriveApp.getFileById(form.getId()).moveTo(folder);
    } catch (e) {
      // フォーム自体は作成済みなので処理は止めず、理由だけ持っておいて後で表示する
      folderMoveError = (e && e.message) ? e.message : String(e);
    }

    step = 'URLの取得';
    const result = {
      editUrl: form.getEditUrl(),
      viewUrl: form.getPublishedUrl(),
      title: title,
      folderMoveError: folderMoveError
    };

    return result;

  } catch (e) {
    // どの段階で失敗したかをそのままエラーメッセージに含めて返す
    throw new Error('「' + step + '」の処理で失敗しました： ' + (e && e.message ? e.message : e));
  }
}

function escapeAttr(s){
  return (s || '').replace(/&/g,'&amp;').replace(/"/g,'&quot;');
}

function renderPerformerList(){
  const wrap = document.getElementById('performerList');
  wrap.innerHTML = '';
  performers.forEach((p, i) => {
    const row = document.createElement('div');
    row.className = 'performer-row';
    row.innerHTML =
      '<div class="order">' + (i + 1) + '</div>' +
      '<input type="text" placeholder="https://youtu.be/..." data-field="url" value="' + escapeAttr(p.url) + '">' +
      '<div class="name-cell">' +
        '<input type="text" placeholder="名前（自動取得）" data-field="name" value="' + escapeAttr(p.name) + '">' +
        '<div class="name-status" data-role="status"></div>' +
      '</div>' +
      '<button class="icon-btn" data-action="remove" title="削除">×</button>';

    const urlInput = row.querySelector('[data-field=url]');
    const nameInput = row.querySelector('[data-field=name]');
    const statusEl = row.querySelector('[data-role=status]');

    renderRowStatus(statusEl, p);

    urlInput.addEventListener('input', () => { p.url = urlInput.value; });
    urlInput.addEventListener('blur', () => {
      const url = urlInput.value.trim();
      if (!url || url === p.lastFetchedUrl) return;
      fetchTitleForRow(p, url, nameInput, statusEl);
    });
    nameInput.addEventListener('input', () => {
      p.name = nameInput.value;
      p.nameEditedByUser = true;
    });
    row.querySelector('[data-action=remove]').addEventListener('click', () => {
      performers = performers.filter(x => x.id !== p.id);
      renderPerformerList();
    });
    wrap.appendChild(row);
  });
}

function renderRowStatus(statusEl, p){
  if (p.status === 'loading'){
    statusEl.classList.remove('error');
    statusEl.innerHTML = '<span class="mini-spinner"></span> タイトルを取得中…';
  } else if (p.status === 'error'){
    statusEl.classList.add('error');
    statusEl.textContent = '取得できませんでした。名前を直接入力してください';
  } else if (p.status === 'done'){
    statusEl.classList.remove('error');
    statusEl.textContent = 'YouTubeのタイトルから取得しました';
  } else {
    statusEl.classList.remove('error');
    statusEl.textContent = '';
  }
}

function fetchTitleForRow(p, url, nameInputEl, statusEl){
  p.status = 'loading';
  renderRowStatus(statusEl, p);
  google.script.run
    .withSuccessHandler(function (results) {
      const r = results && results[0];
      p.lastFetchedUrl = url;
      if (r && r.title){
        if (!p.nameEditedByUser || !p.name){
          p.name = r.title;
          nameInputEl.value = r.title;
        }
        p.status = 'done';
      } else {
        p.status = 'error';
      }
      renderRowStatus(statusEl, p);
    })
    .withFailureHandler(function () {
      p.status = 'error';
      renderRowStatus(statusEl, p);
    })
    .fetchYoutubeTitles([url]);
}

let lastAutoDesc = '';

/**
 * 開催日（例：7/20）から、ネタ見せの対象月（翌月）を計算して返す。
 * 例：7/20 → 8　　12/25 → 1（年をまたぐ場合）
 */
function nextMonthFromDate(dateStr){
  const m = (dateStr || '').match(/(\d{1,2})\s*\/\s*(\d{1,2})/);
  if (!m) return null;
  const month = parseInt(m[1], 10);
  if (month < 1 || month > 12) return null;
  return (month % 12) + 1;
}

function buildDefaultDesc(month){
  const label = month ? (month + '月') : '○月';
  return '🔥' + label + 'ライブのネタ見せ🔥\n'

}

function renderTitlePreview(){
  const date = document.getElementById('eventDate').value.trim();
  const title = date ? (date + ' ネタ見せちゃん') : '';
  document.getElementById('formTitle').value = title;

  // 説明文は、前回こちらが自動で入れた内容から変更されていない場合だけ、
  // 開催日の翌月に合わせて自動更新する（自分で書き換えた文章は上書きしない）
  const descEl = document.getElementById('formDesc');
  if (descEl.value === lastAutoDesc) {
    const month = nextMonthFromDate(date);
    const newDesc = buildDefaultDesc(month);
    descEl.value = newDesc;
    lastAutoDesc = newDesc;
  }
}

function parseBulk(text){
  return text.split('\n')
    .map(line => line.trim())
    .filter(Boolean)
    .map(line => {
      const match = line.match(/https?:\/\/\S+/);
      return match ? match[0] : line;
    })
    .filter(Boolean);
}

/* 開催日欄（例：6/30、12/3）を告知用の4桁表記（0630、1203）に変換する */
function dateToCompact(dateStr){
  const m = (dateStr || '').match(/(\d{1,2})\s*\/\s*(\d{1,2})/);
  if (!m) return (dateStr || '').trim();
  const mm = ('0' + m[1]).slice(-2);
  const dd = ('0' + m[2]).slice(-2);
  return mm + dd;
}
